## Usage

```tsx
import { DataList } from "chakra-ui-solid";
```

```tsx
<DataList.Root>
  <For each={data}>
    {(item) => (
      <DataList.Item>
        <DataList.ItemLabel>{item.label}</DataList.ItemLabel>
        <DataList.ItemValue>{item.value}</DataList.ItemValue>
      </DataList.Item>
    )}
  </For>
</DataList.Root>
```
