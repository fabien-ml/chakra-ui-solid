## Usage

```tsx
import { Group } from "chakra-ui-solid";
```

```tsx
<Group>
  <div />
  <div />
</Group>
```

### Attached

Use the `attached` prop to attach the children together.

<Example name="group-with-attached" />

**Note:** When composing custom components and attaching them to a `Group`, ensure you forward
props.

```tsx
export const Demo = () => {
  return (
    <Group attached>
      <FocusButton />
      <IconButton variant="outline">Two</IconButton>
    </Group>
  );
};

function FocusButton(props: ButtonProps) {
  return <IconButton variant="outline" {...props} />;
}
```

### Out of flow

Mark a child with `data-group-skip` to take it out of the row. It gets no seam of its own and does
not count towards the positions of the children around it, so the two buttons below stay attached to
each other.

```tsx
<Group attached position="relative">
  <Button variant="outline">One</Button>
  <Button variant="outline">Two</Button>
  <Box data-group-skip position="absolute" top="-2" insetEnd="-2">
    <Badge>New</Badge>
  </Box>
</Group>
```
